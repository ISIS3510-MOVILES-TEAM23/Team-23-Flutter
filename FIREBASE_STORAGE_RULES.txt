rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    
    // Reglas existentes para productos
    match /public/products/{userId}/{productId}/{imageId} {
      allow write: if request.auth != null && request.auth.uid == userId;
      allow read: if true;
    }
    
    // NUEVAS REGLAS para feedback
    match /public/feedback/{userId}/{feedbackId}/{imageId} {
      // Permitir subir solo si es tu propio feedback
      allow write: if request.auth != null && request.auth.uid == userId;
      
      // Permitir leer cualquier imagen de feedback (son públicas)
      allow read: if true;
    }
  }
}
